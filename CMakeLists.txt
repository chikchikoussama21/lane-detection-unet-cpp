cmake_minimum_required(VERSION 3.18)
project(lane_cnn_psu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LANE_WITH_TORCH "Construire l'entrainement et l'inference LibTorch CPU" ON)

# inspect_psu ne depend que de C++17: peut etre compile sans OpenCV.
add_executable(inspect_psu src/tools/inspect_psu.cpp)
target_include_directories(inspect_psu PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio highgui)
add_library(lane_opencv INTERFACE)
target_include_directories(lane_opencv INTERFACE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(lane_opencv INTERFACE ${OpenCV_LIBS})
foreach(tool extract_frames audit_dataset prepare_psu make_smoke_psu)
    add_executable(${tool} src/tools/${tool}.cpp)
    target_link_libraries(${tool} PRIVATE lane_opencv)
    target_include_directories(${tool} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
endforeach()

if(LANE_WITH_TORCH)
    find_package(Torch REQUIRED)
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
    add_library(lane_ml src/dataset.cpp src/unet.cpp)
    target_include_directories(lane_ml PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_link_libraries(lane_ml PUBLIC lane_opencv ${TORCH_LIBRARIES})
    foreach(tool train evaluate infer_video hello_libtorch)
        add_executable(${tool} src/${tool}.cpp)
        target_link_libraries(${tool} PRIVATE lane_ml)
    endforeach()
endif()
